 13    router.post('/auth/login', asyncRoute(async (req, res) => {
 14      const { email, password } = validate(z.object({ email: z.email().max(254).transform(value => value.toLowerCase()), password: z.string().min(1).max(128) }).strict(), req.body);
 15      const timestamp = Date.now();
 16      const key = req.ip;
 17      // La ventana y el límite pertenecen a esta instancia local. No es un limitador distribuido.
 18      for (const [ip, item] of attempts) if (item.until < timestamp) attempts.delete(ip);
 19      if (rateLimitEnabled) {
 20        const item = attempts.get(key) || { count: 0, until: timestamp + 15 * 60 * 1000 };
 21        if (item.count >= 10) { res.set('Retry-After', String(Math.max(1, Math.ceil((item.until - timestamp) / 1000)))); throw new HttpError(429, 'Demasiados intentos. Espere 15 minutos e inténtelo de nuevo.'); }
 22        item.count += 1;
 23        attempts.set(key, item);
 24      }
 25      const row = db.prepare('SELECT id,password_hash,active FROM users WHERE email=?').get(email);
 26      const correct = await verifyPassword(password, row?.password_hash || await dummyHash);
 27      if (!correct || !row?.active) throw new HttpError(401, 'Correo o contraseña incorrectos.');
 28      const token = randomToken();
 29      const csrfToken = randomToken();
 30      const expires = new Date(timestamp + 8 * 60 * 60 * 1000);
 31      transaction(db, () => {
 32        db.prepare('DELETE FROM sessions_auth WHERE expires_at<=?').run(now());
 33        if (req.session) db.prepare('DELETE FROM sessions_auth WHERE token_hash=?').run(req.session.token_hash);
 34        db.prepare('INSERT INTO sessions_auth (token_hash,user_id,csrf_token,expires_at) VALUES (?,?,?,?)').run(tokenHash(token), row.id, csrfToken, expires.toISOString());
 35        audit(db, row.id, 'login', 'user', row.id);
 36      });
 37      res.cookie('sgc_session', token, { ...cookieOptions, maxAge: 8 * 60 * 60 * 1000 });
 38      res.json({ data: { user: hydrateUser(db, row.id), csrfToken } });
 39    }));
