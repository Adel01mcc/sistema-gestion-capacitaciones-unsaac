 30  export function createTraining(db, input, user) {
 31    const value = parseTrainingInput(input);
 32    if (!['planificada', 'programada'].includes(value.status)) throw new HttpError(422, 'Una capacitación nueva debe estar planificada o programada.', { status: 'Seleccione planificada o programada.' });
 33    return transaction(db, () => {
 34      checkInstructor(db, value.instructor_id);
 35      assertCode(db, value.code);
 36      const id = insertTraining(db, value, user.id, now());
 37      audit(db, user.id, 'create', 'training', id);
 38      return getTraining(db, id, user);
 39    });
 40  }
 41  export function updateTraining(db, id, input, user) {
 42    const value = parseTrainingInput(input);
 43    return transaction(db, () => {
 44      const current = getTraining(db, id, user);
 45      if (['finalizada', 'cancelada'].includes(current.status)) throw new HttpError(409, 'Una capacitación finalizada o cancelada no admite cambios.');
 46      if (value.status !== current.status && !TRANSITIONS[current.status].includes(value.status)) throw new HttpError(409, 'El cambio de estado solicitado no está permitido.', { status: 'Revise las transiciones documentadas.' });
 47      checkInstructor(db, value.instructor_id);
 48      assertCode(db, value.code, id);
 49      saveTraining(db, id, value, now());
 50      audit(db, user.id, 'update', 'training', id);
 51      return getTraining(db, id, user);
 52    });
 53  }
