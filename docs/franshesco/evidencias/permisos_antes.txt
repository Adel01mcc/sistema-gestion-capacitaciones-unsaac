103  export function updateRole(db, id, input, actorId) {
104    const validPermissions = db.prepare('SELECT code FROM permissions').all().map(row => row.code);
105    const value = validate(z.object({ name: z.string().trim().min(1).max(80), permissions: z.array(z.enum(validPermissions)).max(validPermissions.length).refine(items => new Set(items).size === items.length, 'No repita permisos.') }).strict(), input);
106    return transaction(db, () => {
107      const role = db.prepare('SELECT * FROM roles WHERE id=?').get(id);
108      if (!role) throw new HttpError(404, 'El rol no existe.');
109      if (role.code === 'administrador') throw new HttpError(409, 'El rol administrador está protegido.');
110      db.prepare('UPDATE roles SET name=? WHERE id=?').run(value.name, id);
111      db.prepare('DELETE FROM role_permissions WHERE role_id=?').run(id);
112      for (const permission of value.permissions) db.prepare('INSERT INTO role_permissions (role_id,permission_code) VALUES (?,?)').run(id, permission);
113      audit(db, actorId, 'update_permissions', 'role', id);
114      return listRoles(db).find(item => item.id === id);
115    });
116  }
